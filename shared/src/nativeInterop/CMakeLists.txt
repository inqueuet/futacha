cmake_minimum_required(VERSION 3.21)
project(FutachaNative LANGUAGES C CXX)
add_subdirectory(tracking)
# OpenCV writes generated headers at the top-level build directory.
target_include_directories(futacha_tracking_bridge PRIVATE "${CMAKE_BINARY_DIR}")
add_custom_target(futacha_native DEPENDS futacha_tracking_bridge)
if(ANDROID AND (ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64"))
    add_library(futacha_voicevox_bridge SHARED voicevox/FutachaVoicevox.c voicevox/FutachaVoicevoxJni.c)
    target_include_directories(futacha_voicevox_bridge PRIVATE voicevox "${VOICEVOX_ROOT}/include")
    target_link_libraries(futacha_voicevox_bridge PRIVATE dl)
    target_compile_options(futacha_voicevox_bridge PRIVATE -fvisibility=hidden)
    target_link_options(futacha_voicevox_bridge PRIVATE "-Wl,-z,max-page-size=16384")
    add_dependencies(futacha_native futacha_voicevox_bridge)
endif()
